#include <stdio.h>
#include <stdbool.h>
bool isValid(char* s) {
    int top = -1;
    for (int i = 0; s[i] != '\0'; i++) {
        char current = s[i];
        if (current == '(' || current == '{' || current == '[') {
            top++;
            s[top] = current; 
        } else {
            if (top == -1) {
                return false; 
            }            
            char last_open = s[top];
            if ((current == ')' && last_open == '(') ||
                (current == '}' && last_open == '{') ||
                (current == ']' && last_open == '[')) {
                top--; 
            } else {
                return false; 
            }
        }
    }
    return top == -1;
}
